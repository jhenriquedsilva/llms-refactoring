    private static Set<String> extractTracedTests(CoverageSuitesBundle bundle, final String classFQName, final int lineNumber) {
        Set<String> tests = new HashSet<>();
        final File[] traceFiles = getTraceFiles(bundle);
        if (traceFiles == null) return tests;
        for (File traceFile : traceFiles) {
            try (DataInputStream in = new DataInputStream(new FileInputStream(traceFile))) {
                extractTests(traceFile, in, tests, classFQName, lineNumber);
            }
            catch (Exception ex) {
                LOG.error(traceFile.getName(), ex);
            }
        }
        return tests;
    }

    private static void extractTests(final File traceFile,
                                     final DataInputStream in,
                                     final Set<? super String> tests,
                                     final String classFQName,
                                     final int lineNumber) throws IOException {
        long traceSize = in.readInt();
        for (int i = 0; i < traceSize; i++) {
            final String className = in.readUTF();
            final int linesSize = in.readInt();
            for (int l = 0; l < linesSize; l++) {
                final int line = in.readInt();
                if (Comparing.strEqual(className, classFQName) && lineNumber == line) {
                    tests.add(FileUtilRt.getNameWithoutExtension(traceFile.getName()));
                    return;
                }
            }
        }
    }

    private static File @Nullable [] getTraceFiles(CoverageSuitesBundle bundle) {

            Collections.addAll(files, suiteFiles);
        }
        return files.isEmpty() ? null : files.toArray(new File[0]);

    private boolean checkRootsAndRecompile(@NotNull Module module, @NotNull final CoverageSuitesBundle suite,
                                           @NotNull final Runnable chooseSuiteAction) {
        final CoverageDataManager dataManager = CoverageDataManager.getInstance(module.getProject());
        final boolean includeTests = suite.isTrackTestFolders();
        final VirtualFile[] roots = JavaCoverageClassesEnumerator.getRoots(dataManager, module, includeTests);
        final boolean rootsExist = roots.length >= (includeTests ? 2 : 1) && areRootsValid(roots);

        if (!rootsExist) {
            promptForRecompilation(module, suite);
        }

    private void promptForRecompilation(@NotNull Module module, @NotNull final CoverageSuitesBundle suite) {
        final Project project = module.getProject();
        suite.checkModule(module);

        LOG.debug("Going to ask to rebuild project. Include tests:" + suite.isTrackTestFolders() + ". Module: " + module.getName() + ". Output roots are: ");
        logRoots(JavaCoverageClassesEnumerator.getRoots(CoverageDataManager.getInstance(project), module, suite.isTrackTestFolders()));

        final Notification notification = createRecompilationNotification(project, suite);
        CoverageNotifications.getInstance(project).addNotification(notification);
        notification.notify(project);
    }

    private void logRoots(VirtualFile[] roots) {
        for (VirtualFile root : roots) {

        }
    }

    private Notification createRecompilationNotification(final Project project, final CoverageSuitesBundle suite) {
        final Notification notification = new Notification("Coverage",
                JavaCoverageBundle.message("project.is.out.of.date"),
                JavaCoverageBundle.message("project.class.files.are.out.of.date"),
                NotificationType.INFORMATION);

        notification.addAction(NotificationAction.createSimpleExpiring(JavaCoverageBundle.message("coverage.recompile"), () -> {
            ProjectTaskManagerImpl.putBuildOriginator(project, this.getClass());
            ProjectTaskManager taskManager = ProjectTaskManager.getInstance(project);
            Promise<ProjectTaskManager.Result> promise = taskManager.buildAllModules();
            promise.onSuccess(result -> ApplicationManager.getApplication().invokeLater(() -> {
                CoverageDataManager.getInstance(project).chooseSuitesBundle(suite);
            }, o -> project.isDisposed()));
        }));

        return notification;
    }

    private void collectOutputFiles(@NotNull final PsiFile srcFile, @Nullable final Module module, @NotNull final CoverageSuitesBundle suite, Set<File> classFiles) {
        final CompilerModuleExtension moduleExtension = Objects.requireNonNull(CompilerModuleExtension.getInstance(module));
        final File outputpath = getOutputpath(moduleExtension);
        final File testOutputpath = getTestOutputpath(moduleExtension);

        final VirtualFile outputpathVirtualFile = fileToVirtualFileWithRefresh(outputpath);
        final VirtualFile testOutputpathVirtualFile = fileToVirtualFileWithRefresh(testOutputpath);

        for (JavaCoverageEngineExtension extension : JavaCoverageEngineExtension.EP_NAME.getExtensionList()) {
            if (extension.collectOutputFiles(srcFile, outputpathVirtualFile, testOutputpathVirtualFile, suite, classFiles)) {
                return;
            }
        }

        List<File> children = getAllOutputFiles(srcFile, module, suite);
        PsiClass[] classes = ReadAction.compute(() -> ((PsiClassOwner) srcFile).getClasses());

        collectClassFilesFromChildren(classFiles, children, classes);
    }

    private List<File> getAllOutputFiles(@NotNull final PsiFile srcFile, @Nullable final Module module, @NotNull final CoverageSuitesBundle suite) {
        List<File> children = new ArrayList<>();

        Project project = module.getProject();
        CoverageDataManager dataManager = CoverageDataManager.getInstance(project);
        boolean includeTests = suite.isTrackTestFolders();
        VirtualFile[] roots = JavaCoverageClassesEnumerator.getRoots(dataManager, module, includeTests);

        String packageVmName = AnalysisUtils.fqnToInternalName(getPackageName(srcFile));

        Arrays.stream(roots)
                .map(root -> root.findFileByRelativePath(packageVmName))
                .filter(Objects::nonNull)
                .map(VfsUtilCore::virtualToIoFile)
                .filter(File::exists)
                .map(File::listFiles)
                .filter(Objects::nonNull)
                .forEach(files -> Collections.addAll(children, files));

        return children;
    }

    private void collectClassFilesFromChildren(Set<File> classFiles, List<File> children, PsiClass[] classes) {
        for (PsiClass psiClass : classes) {
            String className = ReadAction.compute(psiClass::getName);
            if (className == null) continue;

            children.stream()
                    .filter(file -> FileUtilRt.extensionEquals(file.getName(), JavaClassFileType.INSTANCE.getDefaultExtension()))
                    .map(FileUtilRt::getNameWithoutExtension)
                    .filter(childName -> childName.equals(className) || (childName.startsWith(className) && childName.charAt(className.length()) == '$'))
                    .forEach(classFiles::add);
        }
    }

    @Override
    public String generateBriefReport(@NotNull Editor editor,
                                      @NotNull PsiFile psiFile,
                                      int lineNumber,
                                      int startOffset,
                                      int endOffset,
                                      @Nullable LineData lineData) {

        }
        try {
            return generateReport(editor, psiFile, lineNumber, startOffset, endOffset, lineData);
        }
        catch (Exception e) {
            LOG.error(e);
            return CoverageBundle.message("hits.title", lineData.getHits());

    private String generateReport(@NotNull Editor editor,
                                  @NotNull PsiFile psiFile,
                                  int lineNumber,
                                  int startOffset,
                                  int endOffset,
                                  @Nullable LineData lineData) {

        for (JavaCoverageEngineExtension extension : JavaCoverageEngineExtension.EP_NAME.getExtensionList()) {
            String report = extension.generateBriefReport(editor, psiFile, lineNumber, startOffset, endOffset, lineData);
            if (report != null) {
                return report;
            }
        }

        TextRange range = TextRange.create(startOffset, endOffset);
        List<SwitchCoverageExpression> switches = JavaCoveragePsiUtilsKt.getSwitches(psiFile, range);
        List<ConditionCoverageExpression> conditions = JavaCoveragePsiUtilsKt.getConditions(psiFile, range);

        return compileCoverageReport(lineData, switches, conditions);
    }

    private String compileCoverageReport(@Nullable LineData lineData, List<SwitchCoverageExpression> switches, List<ConditionCoverageExpression> conditions) {
        StringBuilder buf = new StringBuilder();
        String defaultResult = CoverageBundle.message("hits.title", lineData.getHits());
        buf.append(defaultResult).append("\n");
        int idx = 0;
        int hits = 0;

        hits += addCoverageData(buf, lineData.getJumps(), conditions, index -> index < conditions.size(), JumpData::getTrueHits);
        hits += addCoverageData(buf, lineData.getSwitches(), switches, index -> index < switches.size(), SwitchData::getTotalHits);

        addUnknownOutcomeIfNecessary(buf, lineData, hits);

        return buf.toString();
    }

    private void addUnknownOutcomeIfNecessary(StringBuilder buf, @Nullable LineData lineData, int hits) {
        if (lineData.getHits() > hits && hits > 0) {
            buf.append(JavaCoverageBundle.message("report.unknown.outcome", lineData.getHits() - hits));
        }
    }

    private <T> int addCoverageData(StringBuilder buf, T[] data, List<?> expressions, java.util.function.Predicate<Integer> isIndexValid, java.util.function.ToIntFunction<T> getTotalHits) {
        int hits = 0;
        int idx = 0;
        for (T dataElement : data) {
            if (!isDataElementValid(buf, expressions, isIndexValid, idx)) {
                return 0;
            }
            hits += appendCoverageData(buf, dataElement, expressions.get(idx++));
        }
        return hits;
    }

    private <T, E> int appendCoverageData(StringBuilder buf, T dataElement, E expression) {
        if (dataElement instanceof JumpData) {
            addJumpDataInfo(buf, (JumpData) dataElement, (ConditionCoverageExpression) expression);
            return ((JumpData) dataElement).getTrueHits() + ((JumpData) dataElement).getFalseHits();
        } else if (dataElement instanceof SwitchData) {
            addSwitchDataInfo(buf, (SwitchData) dataElement, (SwitchCoverageExpression) expression, LineCoverage.FULL);
            return IntStream.of(((SwitchData) dataElement).getHits()).sum() + ((SwitchData) dataElement).getDefaultHits();
        }
        return 0;
    }

    private <T> boolean isDataElementValid(StringBuilder buf, List<?> expressions, java.util.function.Predicate<Integer> isIndexValid, int idx) {
        if (isIndexValid.test(idx)) {
            return true;
        } else {
            LOG.info("Cannot map coverage report data with PSI: there are more data elements in report than in PSI");
            return false;
        }
    }

    private static void addJumpDataInfo(StringBuilder buf, JumpData jumpData, ConditionCoverageExpression expression) {
        buf.append(indent).append(expression.getExpression()).append("\n");
        boolean reverse = expression.isReversed();
        int trueHits = reverse ? jumpData.getFalseHits() : jumpData.getTrueHits();
        buf.append(indent).append(indent).append(PsiKeyword.TRUE).append(" ").append(CoverageBundle.message("hits.message", trueHits)).append("\n");

        int falseHits = reverse ? jumpData.getTrueHits() : jumpData.getFalseHits();
        buf.append(indent).append(indent).append(PsiKeyword.FALSE).append(" ").append(CoverageBundle.message("hits.message", falseHits)).append("\n");
    }

    private static void addSwitchDataInfo(StringBuilder buf, SwitchData switchData, SwitchCoverageExpression expression, int coverageStatus) {
        buf.append(indent).append(expression.getExpression()).append("\n");
        boolean allBranchesHit = true;
        for (int i = 0; i < switchData.getKeys().length; i++) {
            String key = i < expression.getCases().size() ? expression.getCases().get(i) : Integer.toString(switchData.getKeys()[i]);
            int switchHits = switchData.getHits()[i];
            allBranchesHit &= switchHits > 0;
            buf.append(indent).append(indent).append(PsiKeyword.CASE).append(" ").append(key).append(": ").append(switchHits).append("\n");
        }
        int defaultHits = switchData.getDefaultHits();
        boolean defaultCausesLinePartiallyCovered = allBranchesHit && coverageStatus != LineCoverage.FULL;
        if (expression.getHasDefault() || defaultCausesLinePartiallyCovered || defaultHits > 0) {
            buf.append(indent).append(indent).append(PsiKeyword.DEFAULT).append(": ").append(defaultHits).append("\n");
        }
    }

    @Override
    @Nullable
    public String getTestMethodName(@NotNull final PsiElement element,
                                    @NotNull final AbstractTestProxy testProxy) {
        if (element instanceof PsiMethod method) {
            PsiClass aClass = method.getContainingClass();
            if (aClass != null) {
                String qualifiedName = ClassUtil.getJVMClassName(aClass);
                if (qualifiedName != null) {
                    return qualifiedName + "," + CoverageListener.sanitize(method.getName(), qualifiedName.length());
                }
            }
        }
        return testProxy.toString();
    }

    @Override
    @NotNull
    public List<PsiElement> findTestsByNames(String @NotNull [] testNames, @NotNull Project project) {
        final List<PsiElement> elements = new ArrayList<>();
        PsiManager psiManager = PsiManager.getInstance(project);

        Arrays.stream(testNames)
                .map(testName -> testName.split(","))
                .filter(parts -> parts.length > 1)
                .forEach(parts -> collectTestsByName(elements, parts[1], parts[0], psiManager));

        return elements;
    }

    private static void collectTestsByName(List<? super PsiElement> elements,
                                           String testName,
                                           String className,
                                           PsiManager psiManager) {
        ReadAction.run(() -> {
            PsiClass psiClass = ClassUtil.findPsiClass(psiManager, className);
            if (psiClass == null) return;
            TestFramework testFramework = TestFrameworks.detectFramework(psiClass);
            if (testFramework == null) return;
            Arrays.stream(psiClass.getAllMethods())
                    .filter(method -> testFramework.isTestMethod(method) && testName.equals(CoverageListener.sanitize(method.getName(), className.length())))
                    .forEach(elements::add);
        });
    }

    @Override
    public final void generateReport(@NotNull final Project project,
                                     @NotNull final DataContext dataContext,
                                     @NotNull final CoverageSuitesBundle currentSuite) {

        final ExportToHTMLSettings settings = ExportToHTMLSettings.getInstance(project);
        ProgressManager.getInstance().run(new Task.Backgroundable(project, JavaCoverageBundle.message("generating.coverage.report")) {
            final Exception[] myExceptions = new Exception[1];

            @Override
            public void run(@NotNull final ProgressIndicator indicator) {
                try {
                    ((JavaCoverageRunner)currentSuite.getSuites()[0].getRunner()).generateReport(currentSuite, project);
                }
                catch (IOException e) {
                    LOG.error(e);
                }
                catch (ReportGenerationFailedException e) {
                    myExceptions[0] = e;
                }
            }

            @Override
            public void onSuccess() {
                if (myExceptions[0] != null) {
                    Messages.showErrorDialog(project, myExceptions[0].getMessage(), CommonBundle.getErrorTitle());
                } else if (settings.OPEN_IN_BROWSER) {
                    BrowserUtil.browse(new File(settings.OUTPUT_DIRECTORY, "index.html"));
                }
            }
        });
    }

        return JavaCoverageBundle.message("java.coverage.engine.presentable.text");
    }

    }
