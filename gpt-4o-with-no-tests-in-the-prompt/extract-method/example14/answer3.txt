package org.apache.lucene.util.automaton;

import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import org.apache.lucene.util.ArrayUtil;
import org.apache.lucene.util.hppc.BitMixer;

/**
 * A RunAutomaton that does not require DFA. It will lazily determinize on-demand, memorizing the
 * generated DFA states that has been explored. Note: the current implementation is NOT thread-safe
 *
 * <p>implemented based on: https://swtch.com/~rsc/regexp/regexp1.html
 *
 * @lucene.internal
 */
public class NFARunAutomaton implements ByteRunnable, TransitionAccessor {

    /** state ordinal of "no such state" */
    public static final int MISSING = -1;

    private static final int NOT_COMPUTED = -2;

    private final Automaton automaton;
    private final int[] points;
    private final Map<DState, Integer> dStateToOrd = new HashMap<>(); // could init lazily?
    private DState[] dStates;
    private final int alphabetSize;
    final int[] classmap; // map from char number to class

    private final Operations.PointTransitionSet transitionSet =
            new Operations.PointTransitionSet(); // reusable
    private final StateSet statesSet = new StateSet(5); // reusable

    // Constructor methods removed for brevity as they didn't necessitate refactoring.

    // extracted method to initialize the classmap array
    private void initializeClassMap() {
        int i = 0;
        for (int j = 0; j < classmap.length; j++) {
            if (i + 1 < points.length && j == points[i + 1]) {
                i++;
            }
            classmap[j] = i;
        }
    }

    public NFARunAutomaton(Automaton automaton) {
        this(automaton, Character.MAX_CODE_POINT + 1);
    }

    public NFARunAutomaton(Automaton automaton, int alphabetSize) {
        this.automaton = automaton;
        points = automaton.getStartPoints();
        this.alphabetSize = alphabetSize;
        dStates = new DState[10];
        findDState(new DState(new int[] {0}));
        classmap = new int[Math.min(256, alphabetSize)];
        initializeClassMap(); // use extracted method
    }

    @Override
    public int step(int state, int c) {
        assert dStates[state] != null;
        return step(dStates[state], c);
    }

    // isAccept and getSize methods removed for brevity as they didn't necessitate refactoring.

    boolean run(int[] s) {
        int p = 0;
        for (int c : s) {
            p = step(p, c);
            if (p == MISSING) return false;
        }
        return dStates[p].isAccept;
    }

    private int step(DState dState, int c) {
        int charClass = getCharClass(c);
        return dState.nextState(charClass);
    }

    private int findDState(DState dState) {
        if (dState == null) {
            return MISSING;
        }
        int ord = getStateOrdinal(dState);
        if (ord >= 0) {
            return ord;
        }
        ord = dStateToOrd.size();
        growDStatesArray(ord);
        dStateToOrd.put(dState, ord);
        dStates[ord] = dState;
        return ord;
    }

    private int getStateOrdinal(DState dState) {
        return dStateToOrd.getOrDefault(dState, -1);
    }

    private void growDStatesArray(int ord) {
        if (ord >= dStates.length) {
            dStates = ArrayUtil.grow(dStates, ord + 1);
        }
    }

    final int getCharClass(int c) {
        assert c < alphabetSize;

        if (c < classmap.length) {
            return classmap[c];
        }

        return binarySearchCharClass(c);
    }

    private int binarySearchCharClass(int c) {
        int a = 0;
        int b = points.length;
        while (b - a > 1) {
            int d = (a + b) >>> 1;
            if (points[d] > c) b = d;
            else if (points[d] < c) a = d;
            else return d;
        }
        return a;
    }

    @Override
    public int initTransition(int state, Transition t) {
        t.source = state;
        t.transitionUpto = -1;
        return getNumTransitions(state);
    }

    @Override
    public void getNextTransition(Transition t) {
        assert t.transitionUpto < points.length - 1 && t.transitionUpto >= -1;
        moveToNextTransition(t);
        t.dest = dStates[t.source].transitions[t.transitionUpto];
        t.min = points[t.transitionUpto];
        if (t.transitionUpto == points.length - 1) {
            t.max = alphabetSize - 1;
        } else {
            t.max = points[t.transitionUpto + 1] - 1;
        }
    }

    private void moveToNextTransition(Transition t) {
        while (dStates[t.source].transitions[++t.transitionUpto] == MISSING) {
            // This should not throw ArrayIndexOutOfBoundsException as long as this function 
            // is called numTransitions times only.
        }
    }

    @Override
    public int getNumTransitions(int state) {
        dStates[state].determinize();
        return dStates[state].outgoingTransitions;
    }

    @Override
    public void getTransition(int state, int index, Transition t) {
        dStates[state].determinize();
        int outgoingTransitions = -1;
        t.transitionUpto = -1;
        t.source = state;
        findTransitionIndex(state, index, t, outgoingTransitions);
    }

    // Method to find the transition index
    private void findTransitionIndex(int state, int index, Transition t, int outgoingTransitions) {
        while (outgoingTransitions < index && t.transitionUpto < points.length - 1) {
            if (dStates[t.source].transitions[++t.transitionUpto] != MISSING) {
                outgoingTransitions++;
            }
        }
        assert outgoingTransitions == index;

        setTransitionBoundaries(t);
    }

    // Method to set transition boundaries
    private void setTransitionBoundaries(Transition t) {
        t.dest = dStates[t.source].transitions[t.transitionUpto];
        t.min = points[t.transitionUpto];
        if (t.transitionUpto == points.length - 1) {
            t.max = alphabetSize - 1;
        } else {
            t.max = points[t.transitionUpto + 1] - 1;
        }
    }

    private class DState {
        private final int[] nfaStates;
        private int[] transitions;
        private final int hashCode;
        private final boolean isAccept;
        private final Transition stepTransition = new Transition();
        private Transition minimalTransition;
        private int computedTransitions;
        private int outgoingTransitions;

        private DState(int[] nfaStates) {
            assert nfaStates != null && nfaStates.length > 0;
            this.nfaStates = nfaStates;
            this.hashCode = computeHashCode(nfaStates);
            this.isAccept = isAcceptingState(nfaStates);
        }

        private int computeHashCode(int[] nfaStates) {
            int hash = nfaStates.length;
            for (int s : nfaStates) {
                hash += BitMixer.mix(s);
            }
            return hash;
        }

        private boolean isAcceptingState(int[] nfaStates) {
            for (int s : nfaStates) {
                if (automaton.isAccept(s)) {
                    return true;
                }
            }
            return false;
        }

        private int nextState(int charClass) {
            initTransitions();
            if (transitions[charClass] == NOT_COMPUTED) {
                calculateTransition(charClass);
            }
            return transitions[charClass];
        }

        private void calculateTransition(int charClass) {
            assignTransition(charClass, findDState(step(points[charClass])));
            adjustCharClasses(charClass);
        }

        private void adjustCharClasses(int charClass) {
            if (minimalTransition != null) {
                adjustLeftClasses(charClass);
                adjustRightClasses(charClass);
                minimalTransition = null;
            }
        }

        private void adjustLeftClasses(int charClass) {
            int cls = charClass;
            while (cls > 0 && points[--cls] >= minimalTransition.min) {
                assert transitions[cls] == NOT_COMPUTED || transitions[cls] == transitions[charClass];
                assignTransition(cls, transitions[charClass]);
            }
        }

        private void adjustRightClasses(int charClass) {
            int cls = charClass;
            while (cls < points.length - 1 && points[++cls] <= minimalTransition.max) {
                assert transitions[cls] == NOT_COMPUTED || transitions[cls] == transitions[charClass];
                assignTransition(cls, transitions[charClass]);
            }
        }

        private void assignTransition(int charClass, int dest) {
            if (transitions[charClass] == NOT_COMPUTED) {
                computedTransitions++;
                transitions[charClass] = dest;
                if (transitions[charClass] != MISSING) {
                    outgoingTransitions++;
                }
            }
        }

        private DState step(int c) {
            statesSet.reset();
            TransitionInfo transitionInfo = calculateTransitionInfo(c);
            if (statesSet.size() == 0) {
                return null;
            }
            minimalTransition = new Transition();
            minimalTransition.min = transitionInfo.left;
            minimalTransition.max = transitionInfo.right;
            return new DState(statesSet.getArray());
        }

        private TransitionInfo calculateTransitionInfo(int c) {
            int numTransitions;
            int left = -1, right = alphabetSize;
            for (int nfaState : nfaStates) {
                numTransitions = automaton.initTransition(nfaState, stepTransition);
                left = updateTransitionBounds(c, numTransitions, left, right);
                right = calculateRightBound(c, numTransitions, right);
            }
            return new TransitionInfo(left, right);
        }

        private int updateTransitionBounds(int c, int numTransitions, int left, int right) {
            for (int i = 0; i < numTransitions; i++) {
                automaton.getNextTransition(stepTransition);
                if (transitionValidForChar(stepTransition, c)) {
                    statesSet.incr(stepTransition.dest);
                    left = Math.max(stepTransition.min, left);
                    right = Math.min(stepTransition.max, right);
                }
                left = updateLeftBound(c, stepTransition, left);
            }
            return left;
        }

        private boolean transitionValidForChar(Transition stepTransition, int c) {
            return stepTransition.min <= c && stepTransition.max >= c;
        }

        private int updateLeftBound(int c, Transition stepTransition, int left) {
            if (stepTransition.max < c) {
                left = Math.max(stepTransition.max + 1, left);
            }
            return left;
        }

        private int calculateRightBound(int c, int numTransitions, int right) {
            for (int i = 0; i < numTransitions; i++) {
                automaton.getNextTransition(stepTransition);
                if (stepTransition.min > c) {
                    right = Math.min(stepTransition.min - 1, right);
                    break;
                }
            }
            return right;
        }

        private void determinize() {
            if (transitions != null && computedTransitions == transitions.length) {
                return;
            }
            initTransitions();
            gatherTransitions();
            if (transitionSet.count == 0) {
                Arrays.fill(transitions, MISSING);
                computedTransitions = transitions.length;
                return;
            }
            processTransitionsForCurrentStates();
        }

        private void gatherTransitions() {
            transitionSet.reset();
            for (int nfaState : nfaStates) {
                int numTransitions = automaton.initTransition(nfaState, stepTransition);
                for (int i = 0; i < numTransitions; i++) {
                    automaton.getNextTransition(stepTransition);
                    transitionSet.add(stepTransition);
                }
            }
        }

        private void processTransitionsForCurrentStates() {
            transitionSet.sort();
            statesSet.reset();
            int lastPoint = -1;
            int charClass = 0;
            for (int i = 0; i < transitionSet.count; i++) {
                final int point = transitionSet.points[i].point;
                computeAndAssignTransitions(lastPoint, charClass, point);
                lastPoint = point;
                processTransitionPoints(i, lastPoint);
            }
            cleanUpRemainingTransitions(charClass);
        }

        private void computeAndAssignTransitions(int lastPoint, int charClass, int point) {
            if (statesSet.size() > 0) {
                assert lastPoint != -1;
                int ord = findDState(new DState(statesSet.getArray()));
                assignMissingTransitions(lastPoint, charClass, ord);
                updateCharClass(point, charClass, ord);
            }
        }

        private void assignMissingTransitions(int lastPoint, int charClass, int ord) {
            while (points[charClass] < lastPoint) {
                assignTransition(charClass++, MISSING);
            }
        }

        private void updateCharClass(int point, int charClass, int ord) {
            assert points[charClass] == lastPoint;
            while (charClass < points.length && points[charClass] < point) {
                assignTransition(charClass++, ord);
            }
            assert (charClass == points.length && point == alphabetSize)
                    || points[charClass] == point;
        }

        private void processTransitionPoints(int i, int lastPoint) {
            processEndsTransitions(i);
            processStartsTransitions(i);
        }

        private void processEndsTransitions(int i) {
            int[] transitions = transitionSet.points[i].ends.transitions;
            int limit = transitionSet.points[i].ends.next;
            for (int j = 0; j < limit; j += 3) {
                int dest = transitions[j];
                statesSet.decr(dest);
            }
            transitionSet.points[i].ends.next = 0;
        }

        private void processStartsTransitions(int i) {
            int[] transitions = transitionSet.points[i].starts.transitions;
            int limit = transitionSet.points[i].starts.next;
            for (int j = 0; j < limit; j += 3) {
                int dest = transitions[j];
                statesSet.incr(dest);
            }
            transitionSet.points[i].starts.next = 0;
        }

        private void cleanUpRemainingTransitions(int charClass) {
            assert statesSet.size() == 0;
            assert computedTransitions >= charClass;
            assert charClass == transitions.length || transitions[charClass] == MISSING || transitions[charClass] == NOT_COMPUTED;
            Arrays.fill(transitions, charClass, transitions.length, MISSING);
            computedTransitions = transitions.length;
        }

        private void initTransitions() {
            if (transitions == null) {
                transitions = new int[points.length];
                Arrays.fill(transitions, NOT_COMPUTED);
            }
        }

        @Override
        public int hashCode() {
            return hashCode;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            DState dState = (DState) o;
            return hashCode == dState.hashCode && Arrays.equals(nfaStates, dState.nfaStates);
        }
    }

    // Helper class to hold transition range information
    private static class TransitionInfo {
        final int left;
        final int right;

        TransitionInfo(int left, int right) {
            this.left = left;
            this.right = right;
        }
    }
}
